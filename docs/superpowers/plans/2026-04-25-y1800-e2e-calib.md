```python
"""Calibration losses: directional Huber + β (Mincer-Zarnowitz) regularizer.

These are designed to be added as light auxiliary terms to a primary
quantile loss. The goal is end-to-end magnitude calibration so that
live trading can use ŷ directly without post-hoc β scaling.

References:
- Mincer & Zarnowitz 1969 (forecast evaluation regression)
- Huber 1964 (robust regression)
- HRT-RNN training script (directional asymmetry inspired)
"""
from __future__ import annotations

import torch


def directional_huber_loss(
    pred: torch.Tensor,
    target: torch.Tensor,
    delta: float = 2.0,
    w_wrong: float = 2.0,
    w_extreme: float = 3.0,
) -> torch.Tensor:
    """Three-tier sign-aware Huber loss.

    Tier 1 (same sign): standard Huber.
    Tier 2 (wrong sign): standard Huber × (1 + w_wrong).
    Tier 3 (wrong sign AND |target| > 1.5): standard Huber × (1 + w_wrong + w_extreme).

    Use when you want magnitude calibration AND directional accuracy.

    Parameters
    ----------
    pred : (N,) predicted values (e.g. q50 from quantile head).
    target : (N,) realized values.
    delta : Huber transition point.
    w_wrong : extra weight on wrong-sign samples.
    w_extreme : additional extra weight when wrong-sign happens on |target|>1.5σ.

    Returns
    -------
    Scalar loss (mean across samples).
    """
    err = pred - target
    abs_err = err.abs()
    base = torch.where(
        abs_err < delta,
        0.5 * err.pow(2),
        delta * (abs_err - 0.5 * delta),
    )
    sign_disagree = (pred.sign() * target.sign() < 0).to(pred.dtype)
    extreme_miss = sign_disagree * (target.abs() > 1.5).to(pred.dtype)
    weight = 1.0 + w_wrong * sign_disagree + w_extreme * extreme_miss
    return (base * weight).mean()


def beta_calib_loss(pred: torch.Tensor, target: torch.Tensor, eps: float = 1e-8) -> torch.Tensor:
    """Penalize batch-level Mincer-Zarnowitz β deviating from 1.

    β = cov(pred, target) / var(pred). β=1 ↔ σ_ŷ = ρ·σ_y (perfect magnitude
    calibration). Any deviation is squared-penalized.

    Use as a light regularizer (typical λ ≤ 0.1) alongside primary loss.
    Batch size should be ≥ 256 for stable β estimation.

    Parameters
    ----------
    pred : (N,) predicted values.
    target : (N,) realized values.
    eps : numerical floor for var(pred).

    Returns
    -------
    Scalar (β - 1)².
    """
    pc = pred - pred.mean()
    tc = target - target.mean()
    cov = (pc * tc).mean()
    var_p = (pc * pc).mean() + eps
    beta = cov / var_p
    return (beta - 1.0).pow(2)
```

```python
import numpy as np
from pathlib import Path
from scipy.stats import pearsonr, spearmanr


def load_pooled(exp_dir: Path):
    yps, ys = [], []
    for f in range(3):
        d = np.load(exp_dir / f"fold_{f}" / "test_preds.npz")
        yp = d["predictions"][:, 1]
        y = d["targets"].squeeze()
        m = d["mask"].astype(bool).squeeze()
        yps.append(yp[m]); ys.append(y[m])
    return np.concatenate(yps), np.concatenate(ys)


def metrics(yp, y, label, ref_p=None, ref_b=None):
    rho = pearsonr(yp, y)[0]
    sp = spearmanr(yp, y)[0]
    beta = np.cov(yp, y, ddof=0)[0, 1] / np.var(yp)
    sr = yp.std() / y.std()
    print(f"\n=== {label} (N={len(y)}) ===")
    print(f"  Pearson    = {rho:+.4f}" + (f"  Δvs ref {rho-ref_p:+.4f}" if ref_p else ""))
    print(f"  Spearman   = {sp:+.4f}")
    print(f"  β          = {beta:+.4f}" + (f"  Δvs ref {beta-ref_b:+.4f}" if ref_b else ""))
    print(f"  σ_ŷ/σ_y    = {sr:.4f}")
    return rho, sp, beta, sr
```

Write to `scripts/y1800_ridge_gate.py`:

```python
"""Ridge walk-forward gate on y_1800.

Two stages:
  Stage 1 (--feats v4): Ridge with V4 64 handcrafted features only.
  Stage 2 (--feats v4_lc): Ridge with V4 64 + 6 long-context (from npz_v4_smooth overlay).

Output: per-fold + pooled Pearson/Spearman.
Gate: pooled clean Pearson >= 0.012 to proceed to V4 DL training.
"""
from __future__ import annotations
import argparse
from pathlib import Path
import numpy as np
from scipy.stats import pearsonr, spearmanr
from sklearn.linear_model import Ridge


def load_features(npz_dir: Path, day_str: str, feat_mode: str):
    f = npz_dir / f"{day_str}.npz"
    if not f.exists():
        return None
    d = np.load(f, allow_pickle=True)
    X = d["X"]
    y = d.get("y_1800")
    m = d.get("y_mask_1800")
    if y is None or m is None:
        return None
    # Use last timestep features (Ridge baseline convention)
    X_last = X[:, -1, :]
    y = np.asarray(y).ravel()
    m = np.asarray(m).astype(bool).ravel()
    if feat_mode == "v4":
        return X_last, y, m
    elif feat_mode == "v4_lc":
        # Try to attach long_context_feats from overlay
        ov = Path("data/npz_v4_smooth") / f"{day_str}.npz"
        if not ov.exists():
            return X_last, y, m  # fallback to v4-only
        d2 = np.load(ov, allow_pickle=True)
        lc = d2.get("long_context_feats")
        if lc is None or lc.shape[0] != X_last.shape[0]:
            return X_last, y, m
        X_full = np.concatenate([X_last, lc.astype(np.float32)], axis=1)
        return X_full, y, m
    else:
        raise ValueError(f"unknown feat_mode {feat_mode}")


def walk_forward_folds(days, train_days=700, val_days=30, test_days=90, stride=60):
    folds = []
    n = len(days)
    start = 0
    while start + train_days + val_days + test_days <= n:
        tr = days[start:start + train_days]
        va = days[start + train_days:start + train_days + val_days]
        te = days[start + train_days + val_days:start + train_days + val_days + test_days]
        folds.append((tr, va, te))
        start += stride
    return folds


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--npz-dir", type=Path, default=Path("data/npz_v4_y1800"))
    ap.add_argument("--feats", choices=["v4", "v4_lc"], default="v4")
    args = ap.parse_args()

    days = sorted(p.stem for p in args.npz_dir.glob("*.npz") if p.stem != "BUILD_INFO")
    print(f"Total days: {len(days)}")

    folds = walk_forward_folds(days)
    print(f"Folds built: {len(folds)}")

    pooled_yp, pooled_y = [], []
    for fi, (tr, va, te) in enumerate(folds):
        Xs_tr, ys_tr, ms_tr = [], [], []
        for d in tr:
            r = load_features(args.npz_dir, d, args.feats)
            if r is None: continue
            X, y, m = r
            Xs_tr.append(X[m]); ys_tr.append(y[m])
        Xs_te, ys_te, ms_te = [], [], []
        for d in te:
            r = load_features(args.npz_dir, d, args.feats)
            if r is None: continue
            X, y, m = r
            Xs_te.append(X[m]); ys_te.append(y[m])
        if not Xs_tr or not Xs_te:
            print(f"Fold {fi}: empty, skip")
            continue
        Xtr = np.concatenate(Xs_tr); ytr = np.concatenate(ys_tr)
        Xte = np.concatenate(Xs_te); yte = np.concatenate(ys_te)
        # Standardize
        mu, sd = Xtr.mean(0), Xtr.std(0) + 1e-9
        Xtr = (Xtr - mu) / sd; Xte = (Xte - mu) / sd
        # Ridge
        model = Ridge(alpha=10.0).fit(Xtr, ytr)
        yp = model.predict(Xte)
        rho = pearsonr(yp, yte)[0]
        sp = spearmanr(yp, yte)[0]
        # Clean stride-10
        ix = np.arange(0, len(yp), 10)
        rho_c = pearsonr(yp[ix], yte[ix])[0]
        sp_c = spearmanr(yp[ix], yte[ix])[0]
        print(f"Fold {fi}: train={len(Xtr)} test={len(Xte)} "
              f"| dense P={rho:+.4f} S={sp:+.4f} | clean P={rho_c:+.4f} S={sp_c:+.4f}")
        pooled_yp.append(yp); pooled_y.append(yte)

    if pooled_yp:
        yp_all = np.concatenate(pooled_yp); y_all = np.concatenate(pooled_y)
        rho_d = pearsonr(yp_all, y_all)[0]; sp_d = spearmanr(yp_all, y_all)[0]
        ix = np.arange(0, len(yp_all), 10)
        rho_c = pearsonr(yp_all[ix], y_all[ix])[0]
        sp_c = spearmanr(yp_all[ix], y_all[ix])[0]
        print(f"\nPOOLED ({args.feats}, N_clean={len(ix)}): "
              f"clean P={rho_c:+.4f} S={sp_c:+.4f} | dense P={rho_d:+.4f} S={sp_d:+.4f}")
        print(f"Gate (clean P >= 0.012): {'PASS' if rho_c >= 0.012 else 'FAIL'}")


if __name__ == "__main__":
    main()
```

- [ ] **Step 5: Create `src/model/backbones/ema_pool_backbone.py`**

Write:

```python
"""EMA pool backbone: dilated causal conv + exponential moving average pool.

EMA over time gives long-context smoothing without learnable parameters
beyond the underlying conv. Decay=0.95 means effective context ~ 1/(1-0.95) = 20 steps.

For y_1800 with input_len=1200, we use decay=0.95 so the EMA's effective
window is ~20 steps (= 20 sec at 1Hz), much longer than the conv's RF=15s but
weighted toward recent.

For longer effective context, use decay=0.99 (effective window 100 steps).
"""
from __future__ import annotations
import torch
import torch.nn as nn

from src.model.temporal_encoder import CausalConv1dBlock


class EMAPoolBackbone(nn.Module):
    """Dilated conv + EMA pool over time."""

    def __init__(self, d_model: int = 32, dropout: float = 0.15, decay: float = 0.95) -> None:
        super().__init__()
        if not (0.0 < decay < 1.0):
            raise ValueError(f"decay must be in (0,1), got {decay}")
        self.decay = float(decay)
        self.temporal_conv = nn.Sequential(
            CausalConv1dBlock(d_model, kernel_size=3, dilation=1, dropout=dropout),
            CausalConv1dBlock(d_model, kernel_size=3, dilation=2, dropout=dropout),
            CausalConv1dBlock(d_model, kernel_size=3, dilation=4, dropout=dropout),
        )

    def forward(self, h: torch.Tensor) -> torch.Tensor:
        # h: (B, L, D)
        h_conv = self.temporal_conv(h)  # (B, L, D)
        # EMA recurrence: ema_t = decay * ema_{t-1} + (1-decay) * h_t
        # Closed-form vectorized: weights are (1-decay) * decay^(L-1-t) for t=0..L-1
        L = h_conv.size(1)
        device = h_conv.device
        # Vectorized weights — most-recent has largest weight
        idx = torch.arange(L - 1, -1, -1, device=device, dtype=h_conv.dtype)
        weights = (1.0 - self.decay) * (self.decay ** idx)
        # Normalize so weights sum to 1 (avoid initial-step underweighting)
        weights = weights / weights.sum()
        # Weighted sum over time
        return (h_conv * weights.view(1, L, 1)).sum(dim=1)
```

```python
"""GRU backbone: dilated causal conv + 1-layer GRU + last hidden state."""
from __future__ import annotations
import torch
import torch.nn as nn

from src.model.temporal_encoder import CausalConv1dBlock


class GRUBackbone(nn.Module):
    def __init__(
        self, d_model: int = 32, hidden: int = 32, n_layers: int = 1, dropout: float = 0.15
    ) -> None:
        super().__init__()
        self.temporal_conv = nn.Sequential(
            CausalConv1dBlock(d_model, kernel_size=3, dilation=1, dropout=dropout),
            CausalConv1dBlock(d_model, kernel_size=3, dilation=2, dropout=dropout),
            CausalConv1dBlock(d_model, kernel_size=3, dilation=4, dropout=dropout),
        )
        self.gru = nn.GRU(
            input_size=d_model,
            hidden_size=hidden,
            num_layers=n_layers,
            batch_first=True,
            dropout=0.0 if n_layers == 1 else dropout,
        )
        # Project hidden back to d_model if different
        self.proj = nn.Identity() if hidden == d_model else nn.Linear(hidden, d_model)

    def forward(self, h: torch.Tensor) -> torch.Tensor:
        # h: (B, L, D)
        h_conv = self.temporal_conv(h)
        out, hN = self.gru(h_conv)  # hN: (n_layers, B, hidden)
        last = hN[-1]               # (B, hidden)
        return self.proj(last)      # (B, D)
```

```python
"""Mamba-2 backbone V2 — clean, isolated, no extra losses or auxiliary modules.

This is the OPPOSITE of V5-LH (which combined Mamba with focal+decorr+side-aware
+cross-path and failed). Here Mamba ONLY replaces the conv stack; everything
else in V4 stays identical. This allows clean A/B comparison.
"""
from __future__ import annotations
import torch
import torch.nn as nn

try:
    from mamba_ssm import Mamba2  # mamba-ssm 2.x
    _HAS_MAMBA = True
except ImportError:
    _HAS_MAMBA = False


class MambaBackboneV2(nn.Module):
    """1-layer Mamba-2 + last-timestep slice. ~12K params at d_model=32."""

    def __init__(
        self, d_model: int = 32, d_state: int = 16, expand: int = 1, dropout: float = 0.15
    ) -> None:
        super().__init__()
        if not _HAS_MAMBA:
            raise ImportError(
                "mamba-ssm not installed. Run: pip install mamba-ssm causal-conv1d"
            )
        # Single Mamba-2 block, no stack — keep params minimal
        self.mamba = Mamba2(
            d_model=d_model,
            d_state=d_state,
            d_conv=4,
            expand=expand,
            headdim=d_model,  # single head to keep simple
        )
        self.dropout = nn.Dropout(dropout)
        # No additional norms or residuals — keep this minimal

    def forward(self, h: torch.Tensor) -> torch.Tensor:
        # h: (B, L, D)
        h2 = self.mamba(h)        # (B, L, D)
        h2 = self.dropout(h2)
        return h2[:, -1, :]       # last timestep, like ConvLastTSBackbone
```

```python
"""Score y_1800 backbone A/B/C variants and declare winner."""
import numpy as np
from pathlib import Path
from scipy.stats import pearsonr, spearmanr


def pooled_metrics(exp_dir: Path):
    yps, ys = [], []
    for f in range(3):
        fd = exp_dir / f"fold_{f}" / "test_preds.npz"
        if not fd.exists():
            return None
        d = np.load(fd)
        yp = d["predictions"][:, 1]
        y = d["targets"].squeeze()
        m = d["mask"].astype(bool).squeeze()
        yps.append(yp[m]); ys.append(y[m])
    yp = np.concatenate(yps); y = np.concatenate(ys)
    ix = np.arange(0, len(yp), 10)
    rho_d, sp_d = pearsonr(yp, y)[0], spearmanr(yp, y)[0]
    rho_c, sp_c = pearsonr(yp[ix], y[ix])[0], spearmanr(yp[ix], y[ix])[0]
    beta = np.cov(yp, y, ddof=0)[0, 1] / np.var(yp)
    sr = yp.std() / y.std()
    return dict(N_dense=len(yp), N_clean=len(ix),
                P_dense=rho_d, S_dense=sp_d,
                P_clean=rho_c, S_clean=sp_c,
                beta=beta, sigma_ratio=sr)


variants = ["baseline_v4", "abc_ema_pool", "abc_gru", "abc_mamba"]
print(f"{'Variant':<20} {'P_clean':>10} {'S_clean':>10} {'β':>8} {'σ_ŷ/σ_y':>10}")
print("-" * 65)
results = {}
for v in variants:
    r = pooled_metrics(Path(f"experiments/y1800_calib/{v}"))
    if r is None:
        print(f"{v:<20} (missing predictions)")
        continue
    results[v] = r
    print(f"{v:<20} {r['P_clean']:>+10.4f} {r['S_clean']:>+10.4f} "
          f"{r['beta']:>+8.4f} {r['sigma_ratio']:>10.4f}")

# Declare winner: highest composite (P+S)/2 among non-collapsed (σ_ŷ/σ_y > 0.2)
print("\n=== Winner determination ===")
candidates = {k: v for k, v in results.items()
              if k != "baseline_v4" and v["sigma_ratio"] > 0.2}
if not candidates:
    print("No variant survived σ_ŷ/σ_y > 0.2 threshold — V5-LH-style collapse.")
    print("Recommendation: stay with baseline_v4 (no architectural upgrade).")
else:
    winner = max(candidates.items(),
                  key=lambda kv: 0.5 * kv[1]["P_clean"] + 0.5 * kv[1]["S_clean"])
    name, m = winner
    print(f"Winner: {name}")
    print(f"  Δ vs baseline_v4: ΔP={m['P_clean'] - results['baseline_v4']['P_clean']:+.4f}, "
          f"ΔS={m['S_clean'] - results['baseline_v4']['S_clean']:+.4f}")
    if m["P_clean"] - results["baseline_v4"]["P_clean"] >= 0.005:
        print("  Gate (ΔP >= +0.005): PASS — proceed to Phase 6 stateful TBPTT")
    else:
        print("  Gate (ΔP >= +0.005): FAIL — backbone not worth complexity, stay with baseline_v4")
```

```python
"""Cross-sectional IC loss (multi-symbol Pearson rank).

At each timestep t, computes Pearson correlation of (pred, target) ACROSS
the symbols present at t. Returns mean(1 - IC) over timesteps with >= 2
valid symbols. For single-symbol training, returns 0.

Inspired by HRT-RNN training script (masked_mtl_loss).

Usage (multi-asset future work):
    pred:   (T, S, H)  — predictions per (timestep, symbol, horizon)
    target: (T, S, H)
    mask:   (T, S, H)  — 1 where valid

For now (single asset), pred is (T, 1, H) and this loss is a no-op.
"""
from __future__ import annotations
import torch


def cross_sectional_ic_loss(
    pred: torch.Tensor,
    target: torch.Tensor,
    mask: torch.Tensor,
    eps: float = 1e-6,
) -> torch.Tensor:
    """Returns mean over timesteps of (1 - cross-sectional Pearson IC).

    Shape conventions:
      pred, target: (T, S) or (T, S, H). If 3D, computed per-horizon then averaged.
      mask: same shape, 1 where valid.

    When S < 2 at every timestep, returns 0 (no-op for single-symbol).
    """
    if pred.dim() == 2:
        pred = pred.unsqueeze(-1)
        target = target.unsqueeze(-1)
        mask = mask.unsqueeze(-1)
    T, S, H = pred.shape

    if S < 2:
        return torch.zeros((), device=pred.device, dtype=pred.dtype)

    valid = (mask > 0) & torch.isfinite(pred) & torch.isfinite(target)
    v = valid.float()
    count = v.sum(dim=1, keepdim=True)        # (T, 1, H)
    enough = (count.squeeze(1) >= 2.0)        # (T, H)
    if not enough.any():
        return torch.zeros((), device=pred.device, dtype=pred.dtype)

    p0 = torch.where(valid, pred, torch.zeros_like(pred))
    t0 = torch.where(valid, target, torch.zeros_like(target))
    mp = p0.sum(dim=1, keepdim=True) / torch.clamp(count, min=1.0)
    mt = t0.sum(dim=1, keepdim=True) / torch.clamp(count, min=1.0)
    pc = torch.where(valid, pred - mp, torch.zeros_like(pred))
    tc = torch.where(valid, target - mt, torch.zeros_like(target))
    num = (pc * tc).sum(dim=1)                # (T, H)
    den = torch.sqrt(torch.clamp((pc * pc).sum(dim=1) * (tc * tc).sum(dim=1),
                                 min=eps))
    ic = num / den                            # (T, H)
    losses = (1.0 - ic) * enough.float()
    n_ok = enough.float().sum().clamp(min=1.0)
    return losses.sum() / n_ok
```
