+def parse_risk_scale(doc: Dict[str, Any]) -> Tuple[float, Dict[str, Any]]:
+    """The producer's own risk multiplier, 1.0 when it says nothing.  PURE.
+
+    ★ FAIL-SAFE IS 1.0 IN ONE DIRECTION ONLY, AND THAT ASYMMETRY IS THE POINT.
+    ABSENT  -> 1.0, accepted, reason "absent".  An old producer, or a producer running no sleeve,
+               must be bitwise the current behaviour.
+    PRESENT but not on the whitelist / not finite / not a number -> the WHOLE FILE IS REJECTED
+               (see parse_target below), NOT coerced to 1.0.  A producer that tried to say something
+               about risk and said it wrongly is a producer we do not trust to size the book; the
+               existing on_unavailable="hold" path (hold current positions + HIGH alarm) is the
+               honest response, and it is already written, already tested, already pre-registered.
+               Coercing a malformed risk statement to "full risk" is how sigma_ladder's `missing`
+               became invisible.  We do not repeat it.
+    The whitelist is DISCRETE on purpose: a continuous field lets a producer arithmetic bug write
+    0.003 or 4.0 and be believed.  Three values are enough to say "fine / degraded / badly degraded".
+    """
+    if "risk_scale" not in doc:
+        return 1.0, {"risk_scale": 1.0, "declared": False, "accepted": True, "reason": "absent"}
+    v = doc.get("risk_scale")
+    if isinstance(v, bool) or not isinstance(v, (int, float)):
+        return 1.0, {"risk_scale": None, "declared": True, "accepted": False,
+                     "reason": f"risk_scale not a number: {v!r}"}
+    f = float(v)
+    if math.isnan(f) or math.isinf(f):
+        return 1.0, {"risk_scale": None, "declared": True, "accepted": False,
+                     "reason": f"risk_scale not finite: {f}"}
+    if not any(abs(f - a) < 1e-12 for a in _RISK_SCALE_ALLOWED):
+        return 1.0, {"risk_scale": None, "declared": True, "accepted": False,
+                     "reason": f"risk_scale {f} not in whitelist {list(_RISK_SCALE_ALLOWED)}"}
+    rr = doc.get("risk_reason")
+    return f, {"risk_scale": f, "declared": True, "accepted": True, "reason": None,

+               "sleeve_alloc": (float(doc["sleeve_alloc"])
+                                if isinstance(doc.get("sleeve_alloc"), (int, float))
+                                and not isinstance(doc.get("sleeve_alloc"), bool) else None)}
